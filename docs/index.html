Ex. No. 12 – Full Stack Web Application Using Node.js, React.js and MongoDB

React (Vite) front end, Express REST API and MongoDB storage implementing create, read, update and delete for student records.

This is a server-side exercise, so GitHub Pages can only show the source and the output captured from a real local run (Node.js, MongoDB 8.2 and mongosh on the author's machine). The React front end calls http://localhost:5000, so the live app only works when the backend is running locally.

Source code

backend/server.js
require('dotenv').config();
const express = require('express');
const cors = require('cors');
const { MongoClient, ObjectId } = require('mongodb');

const app = express();
const client = new MongoClient(process.env.MONGO_URI);

app.use(cors());
app.use(express.json());

let students;

app.get('/api/students', async (req, res) => {
    try {
        const records = await students.find({}).toArray();
        res.json(records);
    } catch (error) {
        res.status(500).json({ error: 'Unable to fetch students' });
    }
});

app.post('/api/students', async (req, res) => {
    try {
        const { name, department } = req.body;

        if (typeof name !== 'string' || !name.trim() ||
            typeof department !== 'string' || !department.trim()) {
            return res.status(400).json({
                error: 'Name and department are required'
            });
        }

        const student = {
            name: name.trim(),
            department: department.trim()
        };

        const result = await students.insertOne(student);

        res.status(201).json({
            _id: result.insertedId,
            ...student
        });
    } catch (error) {
        res.status(500).json({ error: 'Unable to add student' });
    }
});

app.put('/api/students/:id', async (req, res) => {
    try {
        const { id } = req.params;
        const { name, department } = req.body;

        if (!ObjectId.isValid(id)) {
            return res.status(400).json({ error: 'Invalid student ID' });
        }

        if (typeof name !== 'string' || !name.trim() ||
            typeof department !== 'string' || !department.trim()) {
            return res.status(400).json({
                error: 'Name and department are required'
            });
        }

        const result = await students.updateOne(
            { _id: new ObjectId(id) },
            { $set: {
                name: name.trim(),
                department: department.trim()
            }}
        );

        if (result.matchedCount === 0) {
            return res.status(404).json({ error: 'Student not found' });
        }

        res.json({ message: 'Student updated successfully' });
    } catch (error) {
        res.status(500).json({ error: 'Unable to update student' });
    }
});

app.delete('/api/students/:id', async (req, res) => {
    try {
        const { id } = req.params;

        if (!ObjectId.isValid(id)) {
            return res.status(400).json({ error: 'Invalid student ID' });
        }

        const result = await students.deleteOne({
            _id: new ObjectId(id)
        });

        if (result.deletedCount === 0) {
            return res.status(404).json({ error: 'Student not found' });
        }

        res.json({ message: 'Student deleted successfully' });
    } catch (error) {
        res.status(500).json({ error: 'Unable to delete student' });
    }
});

async function startServer() {
    try {
        await client.connect();

        const db = client.db('CollegeDB');
        students = db.collection('Students');

        app.listen(process.env.PORT || 5000, () => {
            console.log('Backend running on port 5000');
        });
    } catch (error) {
        console.error('Database connection failed:', error.message);
        process.exit(1);
    }
}

startServer();
frontend/src/App.jsx
import { useEffect, useState } from 'react';
import './App.css';

const API = 'http://localhost:5000/api/students';

function App() {
    const [students, setStudents] = useState([]);
    const [name, setName] = useState('');
    const [department, setDepartment] = useState('');
    const [editId, setEditId] = useState(null);
    const [message, setMessage] = useState('');

    async function loadStudents() {
        try {
            const response = await fetch(API);
            if (!response.ok) throw new Error('Unable to load students');
            setStudents(await response.json());
        } catch (error) {
            setMessage(error.message);
        }
    }

    useEffect(() => {
        loadStudents();
    }, []);

    async function handleSubmit(event) {
        event.preventDefault();
        setMessage('');

        try {
            const response = await fetch(
                editId ? `${API}/${editId}` : API,
                {
                    method: editId ? 'PUT' : 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ name, department })
                }
            );

            const data = await response.json();
            if (!response.ok) throw new Error(data.error);

            setName('');
            setDepartment('');
            setEditId(null);
            setMessage(data.message || 'Student added successfully');
            await loadStudents();
        } catch (error) {
            setMessage(error.message || 'Request failed');
        }
    }

    function editStudent(student) {
        setEditId(student._id);
        setName(student.name);
        setDepartment(student.department);
        setMessage('');
    }

    async function deleteStudent(id) {
        if (!window.confirm('Delete this student?')) return;

        try {
            const response = await fetch(`${API}/${id}`, {
                method: 'DELETE'
            });

            const data = await response.json();
            if (!response.ok) throw new Error(data.error);

            setMessage(data.message);
            await loadStudents();
        } catch (error) {
            setMessage(error.message);
        }
    }

    return (
        <main className="container">
            <h1>Student Management System</h1>

            <form onSubmit={handleSubmit}>
                <input
                    placeholder="Student Name"
                    value={name}
                    onChange={e => setName(e.target.value)}
                    required
                />

                <input
                    placeholder="Department"
                    value={department}
                    onChange={e => setDepartment(e.target.value)}
                    required
                />

                <button type="submit">
                    {editId ? 'Update Student' : 'Add Student'}
                </button>

                {editId && (
                    <button
                        type="button"
                        onClick={() => {
                            setEditId(null);
                            setName('');
                            setDepartment('');
                        }}
                    >
                        Cancel
                    </button>
                )}
            </form>

            {message && <p role="status">{message}</p>}

            <h2>Student Records</h2>

            <table>
                <thead>
                    <tr>
                        <th>Name</th>
                        <th>Department</th>
                        <th>Actions</th>
                    </tr>
                </thead>

                <tbody>
                    {students.map(student => (
                        <tr key={student._id}>
                            <td>{student.name}</td>
                            <td>{student.department}</td>
                            <td>
                                <button
                                    onClick={() => editStudent(student)}
                                >
                                    Edit
                                </button>

                                <button
                                    onClick={() => deleteStudent(student._id)}
                                >
                                    Delete
                                </button>
                            </td>
                        </tr>
                    ))}
                </tbody>
            </table>
        </main>
    );
}

export default App;
